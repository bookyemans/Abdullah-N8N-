// ─────────────────────────────────────────
// 🎨 إنشاء ورقة مع التنسيق
// ─────────────────────────────────────────

function createSheet(ss, cfg, index) {
  var sheet = ss.getSheetByName(cfg.name);

  if (!sheet) {
    sheet = ss.insertSheet(cfg.name, index);
  } else {
    sheet.clear();
    sheet.clearFormats();
  }

  // لوحة التحكم تُعالج منفصلة
  if (cfg.headers.length === 0) return;

  // ── لون التبويب ──
  sheet.setTabColor(cfg.color);

  // ── العناوين ──
  var headerRange = sheet.getRange(1, 1, 1, cfg.headers.length);
  headerRange.setValues([cfg.headers]);

  // ── تنسيق العناوين ──
  headerRange
    .setBackground(cfg.color)
    .setFontColor('#FFFFFF')
    .setFontWeight('bold')
    .setFontSize(11)
    .setHorizontalAlignment('center')
    .setVerticalAlignment('middle')
    .setWrap(true);

  sheet.setRowHeight(1, 40);

  // ── عرض الأعمدة ──
  for (var i = 0; i < cfg.widths.length; i++) {
    sheet.setColumnWidth(i + 1, cfg.widths[i]);
  }

  // ── تجميد الصف الأول ──
  sheet.setFrozenRows(1);

  // ── تنسيق منطقة البيانات ──
  var dataRange = sheet.getRange(2, 1, 998, cfg.headers.length);
  dataRange
    .setFontSize(10)
    .setVerticalAlignment('middle')
    .setHorizontalAlignment('center')
    .setWrap(true);

  // ── ألوان صفوف متناوبة (أول 200 صف) ──
  for (var row = 2; row <= 200; row++) {
    var rowRange = sheet.getRange(row, 1, 1, cfg.headers.length);
    rowRange.setBackground(row % 2 === 0 ? '#F0F4F8' : '#FFFFFF');
  }

  // ── فلتر ──
  sheet.getRange(1, 1, 1, cfg.headers.length).createFilter();

  // ── حماية العناوين ──
  var protection = sheet.getRange(1, 1, 1, cfg.headers.length).protect();
  protection.setDescription('حماية العناوين');
  protection.setWarningOnly(true);
}

// ─────────────────────────────────────────
// 🔧 أدوات مساعدة للوحة التحكم
// ─────────────────────────────────────────

function sectionTitle(sheet, row, title, bgColor) {
  sheet.getRange('B' + row + ':G' + row).merge()
    .setValue(title)
    .setBackground(bgColor)
    .setFontSize(13)
    .setFontWeight('bold')
    .setHorizontalAlignment('center');
  sheet.setRowHeight(row, 35);
}

function deleteDefaultSheets(ss) {
  var names = ['Sheet1', 'ورقة1'];
  for (var i = 0; i < names.length; i++) {
    try {
      var s = ss.getSheetByName(names[i]);
      if (s && ss.getSheets().length > 1) ss.deleteSheet(s);
    } catch (e) { }
  }
}


// ═══════════════════════════════════════════════════════════
// 📥 استقبال البيانات من الموقع
// ═══════════════════════════════════════════════════════════

function doPost(e) {
  try {
    var lock = LockService.getScriptLock();
    lock.tryLock(10000);

    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var type = data.type || 'visit';

    if (type === 'email_open') {
      handleEmail(ss, data);
    } else {
      handleVisit(ss, data);
    }

    lock.releaseLock();

    return ContentService.createTextOutput(JSON.stringify({
      status: 'ok',
      type: type
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: 'error',
      message: err.message
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  return ContentService.createTextOutput(JSON.stringify({
    status: 'ok',
    message: 'Smart Business Tracker API v3.0',
    sheets: 3
  })).setMimeType(ContentService.MimeType.JSON);
}


// ─────────────────────────────────────────
// 📄 معالجة الزيارات
// ─────────────────────────────────────────

function handleVisit(ss, data) {
  var sheet = ss.getSheetByName('📊 الزيارات');
  if (!sheet) return;

  var type = data.type || 'visit';

  // ── إذا كان تحديث خروج: حدّث السطر الموجود ──
  if (type === 'visit_exit') {
    updateExitData(sheet, data);
    return;
  }

  // ── زيارة جديدة: أضف سطر ──
  var row = [
    formatDate(data.timestamp),                          // التاريخ
    data.user || 'مجهول',                                // اليوزر
    data.isNew ? 'جديد' : 'عائد',                       // جديد/عائد
    data.visitCount || 1,                                // رقم الزيارة
    data.pageName || getPageName(data.page),              // الصفحة
    data.browsePath || '-',                              // مسار التصفح
    data.pageCount || 1,                                 // عدد الصفحات
    0,                                                   // مدة البقاء (تُحدّث عند الخروج)
    0,                                                   // نسبة القراءة (تُحدّث عند الخروج)
    data.source || 'مباشر',                              // المصدر
    data.sourceType || 'مباشر 🏠',                       // نوع المصدر
    data.campaign || '-',                                // الحملة
    data.country || '-',                                 // الدولة
    data.deviceType || '-',                              // الجهاز
    data.browser || '-'                                  // المتصفح
  ];

  sheet.appendRow(row);

  // ── حفظ رقم السطر للتحديث لاحقاً ──
  // نستخدم sessionId + page كمعرف فريد
}


// ─────────────────────────────────────────
// 🔄 تحديث بيانات الخروج
// ─────────────────────────────────────────

function updateExitData(sheet, data) {
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return;

  var user = data.user || 'مجهول';
  var page = data.pageName || getPageName(data.page);

  // ── البحث من الأسفل (أسرع) عن آخر سطر لهذا اليوزر في هذه الصفحة ──
  var searchRange = sheet.getRange(2, 1, lastRow - 1, 5).getValues();

  for (var i = searchRange.length - 1; i >= 0; i--) {
    var rowUser = String(searchRange[i][1]);  // عمود B = اليوزر
    var rowPage = String(searchRange[i][4]);  // عمود E = الصفحة

    if (rowUser === user && rowPage === page) {
      var actualRow = i + 2;  // +2 لأن البيانات تبدأ من صف 2

      // تحديث مدة البقاء (عمود H = 8)
      sheet.getRange(actualRow, 8).setValue(data.timeOnPage || 0);

      // تحديث نسبة القراءة (عمود I = 9)
      sheet.getRange(actualRow, 9).setValue(data.scrollMax || 0);

      // تحديث مسار التصفح (عمود F = 6)
      if (data.browsePath) {
        sheet.getRange(actualRow, 6).setValue(data.browsePath);
      }

      // تحديث عدد الصفحات (عمود G = 7)
      if (data.pageCount) {
        sheet.getRange(actualRow, 7).setValue(data.pageCount);
      }

      return;
    }
  }

  // ── إذا لم يجد السطر: أضف سطر جديد بالبيانات الكاملة ──
  var row = [
    formatDate(data.timestamp),
    user,
    data.isNew ? 'جديد' : 'عائد',
    data.visitCount || 1,
    page,
    data.browsePath || '-',
    data.pageCount || 1,
    data.timeOnPage || 0,
    data.scrollMax || 0,
    data.source || 'مباشر',
    data.sourceType || 'مباشر 🏠',
    data.campaign || '-',
    data.country || '-',
    data.deviceType || '-',
    data.browser || '-'
  ];

  sheet.appendRow(row);
}

function handleEmail(ss, data) {

    data.emailClient || data.browser || '-', // تطبيق البريد
    data.ip || '-'                         // IP
  ];

  sheet.appendRow(row);
}

function formatDate(d) {
  try {
    var date = new Date(d);
    if (isNaN(date.getTime())) date = new Date();
    return Utilities.formatDate(date, Session.getScriptTimeZone(), 'yyyy/MM/dd HH:mm:ss');
  } catch (e) {
    return Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy/MM/dd HH:mm:ss');
  }
}

function getPageName(path) {
  if (!path) return 'الرئيسية';
  var p = path.toLowerCase();

  if (p.indexOf('/blogs/') !== -1 || p.indexOf('/blog/') !== -1) {
    var articleName = p.split('/').pop().replace('.html', '');
    if (articleName && articleName !== 'blog') return 'مقال: ' + articleName;
  }
  if (p.indexOf('/manager/') !== -1) {
    var projectName = p.split('/').pop().replace('.html', '');
    return 'مشروع: ' + projectName;
  }

  if (p.indexOf('about') !== -1) return 'من أنا';
  if (p.indexOf('service') !== -1) return 'الخدمات';
  if (p.indexOf('project') !== -1) return 'المشاريع';
  if (p.indexOf('blog') !== -1) return 'المدونة';
  if (p.indexOf('contact') !== -1) return 'تواصل';

  return 'الرئيسية';
}

function getEngagementLevel(openNum) {
  if (openNum >= 10) return '💎 عميل مخلص';
  if (openNum >= 5) return '🔥🔥🔥 مهتم جداً';
  if (openNum >= 3) return '🔥🔥 مهتم';
  if (openNum >= 2) return '🔥 فضولي';
  return '🆕 أول فتح';
}

function updateDashboard() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var dash = ss.getSheetByName('📈 لوحة التحكم');
  if (!dash) return;

  updateTopUsers(ss, dash);
  updateTopCountries(ss, dash);
}


function updateTopUsers(ss, dash) {
  var visitSheet = ss.getSheetByName('📊 الزيارات');
  if (!visitSheet || visitSheet.getLastRow() < 2) return;

  var data = visitSheet.getRange(2, 1, visitSheet.getLastRow() - 1, 5).getValues();
  var users = {};

  // عدّ الزيارات لكل يوزر
  for (var i = 0; i < data.length; i++) {
    var u = String(data[i][1]); // عمود B
    if (u && u !== 'مجهول' && u !== '-') {
      if (!users[u]) users[u] = { visits: 0, lastDate: '' };
      users[u].visits++;
      users[u].lastDate = data[i][0] || '';
    }
  }

  // ترتيب
  var sorted = Object.keys(users).sort(function (a, b) {
    return users[b].visits - users[a].visits;
  });

  // عدّ إيميلات
  var emailSheet = ss.getSheetByName('📧 الإيميلات');
  var emailCounts = {};
  if (emailSheet && emailSheet.getLastRow() > 1) {
    var eData = emailSheet.getRange(2, 2, emailSheet.getLastRow() - 1, 1).getValues();
    for (var j = 0; j < eData.length; j++) {
      var eu = String(eData[j][0]);
      if (eu && eu !== '-') emailCounts[eu] = (emailCounts[eu] || 0) + 1;
    }
  }

  // كتابة أنشط 10
  for (var k = 0; k < 10; k++) {
    var row = 28 + k;

    if (k < sorted.length) {
      var name = sorted[k];
      var v = users[name];
      var emails = emailCounts[name] || 0;
      var total = v.visits + emails;
      var rating = total >= 15 ? '🔥🔥🔥' : total >= 8 ? '🔥🔥' : total >= 3 ? '🔥' : '⚡';

      dash.getRange(row, 2).setValue(name);
      dash.getRange(row, 3).setValue(v.visits);
      dash.getRange(row, 4).setValue(emails);
      dash.getRange(row, 5).setValue(v.lastDate);
      dash.getRange(row, 6).setValue('-');
      dash.getRange(row, 7).setValue(rating);
    } else {
      // تنظيف الصفوف الفارغة
      dash.getRange(row, 2, 1, 6).clearContent();
    }
  }
}


function updateTopCountries(ss, dash) {
  var visitSheet = ss.getSheetByName('📊 الزيارات');
  if (!visitSheet || visitSheet.getLastRow() < 2) return;

  var data = visitSheet.getRange(2, 13, visitSheet.getLastRow() - 1, 1).getValues(); // عمود M = الدولة
  var total = data.length;
  var countries = {};

  for (var i = 0; i < data.length; i++) {
    var c = String(data[i][0]);
    if (c && c !== '-' && c !== 'غير معروف') {
      countries[c] = (countries[c] || 0) + 1;
    }
  }

  var sorted = Object.keys(countries).sort(function (a, b) {
    return countries[b] - countries[a];
  });

  for (var k = 0; k < 10; k++) {
    var row = 41 + k;

    if (k < sorted.length) {
      var name = sorted[k];
      var count = countries[name];
      var pct = total > 0 ? Math.round((count / total) * 100 * 10) / 10 : 0;

      dash.getRange(row, 2).setValue(name);
      dash.getRange(row, 3).setValue(count);
      dash.getRange(row, 4).setValue(pct + '%');
    } else {
      dash.getRange(row, 2, 1, 6).clearContent();
    }
  }
}

function createTimeTrigger() {
  // حذف التريجرات القديمة
  var triggers = ScriptApp.getProjectTriggers();
  for (var i = 0; i < triggers.length; i++) {
    if (triggers[i].getHandlerFunction() === 'updateDashboard') {
      ScriptApp.deleteTrigger(triggers[i]);
    }
  }

  // تريجر كل ساعة
  ScriptApp.newTrigger('updateDashboard')
    .timeBased()
    .everyHours(1)
    .create();

  SpreadsheetApp.getUi().alert('✅ تم تفعيل التحديث التلقائي كل ساعة');
}
